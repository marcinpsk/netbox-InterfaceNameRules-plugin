3"""Apply one rule to a batch of installed modules through family plans. 

4 

5Every module in the batch is planned family by family and executed family by family, so a module 

6that cannot take its names costs the batch only that family. The module rows, their interfaces and 

7each module type's templates are read once for the whole batch. 

8"""

31# A member left with the name it had for a reason the operator can act on. An unsupported topology 

32# is not one of them: the release cannot hold the family, so nothing was dropped by this batch. 

33_SKIPPED_STATUSES = (FamilyStatus.BLOCKED, FamilyStatus.STALE, FamilyStatus.FAILED) 

34 

35_EXECUTORS = { 

36 InstalledFamilyPlan: execute_installed_plan, 

37 StructuralFamilyPlan: execute_structural_family, 

38 FlatCreationPlan: execute_flat_family, 

39}

42@dataclass(frozen=True, slots=True) 

43class ModuleFamilyPlans: 

44 """The families a rule intends on one module, split by how each was found. 

45 

46 *installed* are the families the module already carries; *leftover* are the plans for the 

47 interfaces no installed family claimed, whether the rule renames one or builds a family on it. 

48 """ 

49 

50 installed: tuple 

51 leftover: tuple 

52 

53 @property 

54 def plans(self) -> tuple: 

55 """Return every plan, installed families first.""" 

56 return (*self.installed, *self.leftover)

59@dataclass(frozen=True, slots=True) 

60class BatchOutcome: 

61 """Every family one batch operation planned, and what happened to it.""" 

62 

63 families: tuple[FamilyOutcome, ...] 

64 

65 @property 

66 def changed_count(self) -> int: 

67 """Return the number of interfaces the batch renamed or created.""" 

68 return sum(family.changed_count for family in self.families) 

69 

70 @property 

71 def skipped_members(self) -> tuple[MemberOutcome, ...]: 

72 """Return every member a collision, a stale plan or a failure left as it was.""" 

73 return tuple( 

74 member for family in self.families for member in family.members if member.status in _SKIPPED_STATUSES 

75 ) 

76 

77 @property 

78 def changed_families(self) -> tuple[FamilyOutcome, ...]: 

79 """Return every family this batch actually rewrote.""" 

80 return tuple(family for family in self.families if family.status == FamilyStatus.CHANGED) 

81 

82 @property 

83 def blocked_families(self) -> tuple[FamilyOutcome, ...]: 

84 """Return every family a collision, a stale plan or a failure left as it was.""" 

85 return tuple(family for family in self.families if family.status in _SKIPPED_STATUSES)

88def execute_family_plan(plan) -> FamilyOutcome: 

89 """Execute one planned family through the executor its plan kind owns. 

90 

91 A plan carrying no live rows (a prospective plan above all) has no executor, so a preview 

92 object is refused here rather than locking anything. 

93 """ 

94 executor = _EXECUTORS.get(type(plan)) 

95 if executor is None: 

96 raise TypeError(f"{type(plan).__name__} is not an executable family plan") 

97 return executor(plan)

100def _is_channel(interface) -> bool: 

101 """Return whether *interface* is bound to a parent channel.""" 

102 return getattr(interface, "channel_id", None) is not None

105def _creation_plan(module, rule, variables, base): 

106 """Return the plan that builds the family *rule* describes on one plain interface.""" 

107 if builds_channelized_family(rule): 

108 return plan_structural_family(module, rule, variables, base) 

109 return plan_flat_family(module, rule, variables, base)

112def _creation_plans(module, rule, variables, plain): 

113 """Return one creation plan per family, so two bases of one family never build it twice.""" 

114 candidates = [(base, intended_family_names(rule, variables, base.name)) for base in plain] 

115 kept = one_family_per_name_set([(base.name, target_names) for base, target_names in candidates]) 

116 return [_creation_plan(module, rule, variables, candidates[index][0]) for index in kept]

119def plan_module_families(module, rule, variables, interfaces, admit_leftover=None) -> ModuleFamilyPlans: 

120 """Return one executable plan for every family *rule* intends on *module*. 

121 

122 Every interface belongs to at most one plan: an installed family claims its members first, and 

123 what is left over is planned as the family the rule would build on it. 

124 

125 *admit_leftover* filters the interfaces no installed family claimed. It runs before two of 

126 them that intend one family are collapsed into it, so a caller that must not touch one of the 

127 two cannot have it survive the collapse as the row the family is built on. 

128 """ 

129 installed = plan_installed_families(module, rule, variables, interfaces=interfaces) 

130 claimed = installed.member_pks 

131 plain = [interface for interface in interfaces if interface.pk not in claimed and not _is_channel(interface)] 

132 if admit_leftover is not None: 

133 plain = list(admit_leftover(plain)) 

134 if rule.channel_count <= 0: 

135 leftover = tuple(plan_interface_rename(module, rule, variables, interface) for interface in plain) 

136 elif any(plan.topology == FamilyTopology.CHANNELIZED for plan in installed.plans): 

137 # A breakout rule renames the families the module already models; it never adds one beside them. 

138 leftover = () # pragma: no cover - requires channelization support 

139 for interface in plain: # pragma: no cover - see above 

140 logger.debug( 

141 "Interface %r is not channelized; skipping it while rule '%s' breaks out this module's families.", 

142 interface.name, 

143 rule, 

144 ) 

145 else: 

146 leftover = tuple(_creation_plans(module, rule, variables, plain)) 

147 return ModuleFamilyPlans(installed=installed.plans, leftover=leftover)

150def _selection_pks(plan): 

151 """Return the interface primary keys a selection reaches this family through. 

152 

153 A channelized family is submitted through its parent alone, because a channel is not an 

154 independent candidate on any path. 

155 """ 

156 if isinstance(plan, InstalledFamilyPlan): 

157 if plan.parent_pk is None: 

158 return plan.member_pks 

159 return (plan.parent_pk,) # pragma: no cover - requires channelization support 

160 return (plan.base.pk,)

163def _selected(plans, selected_pks): 

164 """Return the plans the operator's interface selection reaches.""" 

165 if selected_pks is None: 

166 return plans 

167 return [plan for plan in plans if selected_pks.intersection(_selection_pks(plan))]

170def execute_module_families(plans): 

171 """Execute each planned family in order.""" 

172 return [execute_family_plan(plan) for plan in plans]

175def _apply_module(rule, module, interfaces, selected_pks): 

176 """Plan and execute every selected family on one module.""" 

177 variables = build_variables(module.module_bay, device=module.device) 

178 plans = _selected(plan_module_families(module, rule, variables, interfaces).plans, selected_pks) 

179 return execute_module_families(plans)

182def apply_rule_to_modules(rule, modules, selected_pks=None, limit=None) -> BatchOutcome: 

183 """Apply *rule* to every module in *modules*, one family at a time. 

184 

185 *selected_pks* limits the batch to the families those interfaces reach; *limit* stops it after 

186 the module that reached that many changed interfaces. 

187 """ 

188 if not modules: 

189 return BatchOutcome(families=()) 

190 by_module = interfaces_by_module(modules) 

191 families: list[FamilyOutcome] = [] 

192 changed = 0 

193 with pinned_template_cache(modules): 

194 for module in modules: 

195 outcomes = _apply_module(rule, module, by_module[module.pk], selected_pks) 

196 families.extend(outcomes) 

197 changed += sum(outcome.changed_count for outcome in outcomes) 

198 if limit is not None and changed >= limit: 

199 break 

200 return BatchOutcome(families=tuple(families))
