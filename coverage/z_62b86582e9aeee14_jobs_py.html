24def run_as_job_user(job, body): 

25 """Call *body*, which takes no arguments, as a request of the user who enqueued *job*. 

26 

27 NetBox writes a change log only while a request is current, and a worker has none. The request 

28 carries the job ID, so the change log lists the changes of one job under one request ID. 

29 """ 

30 if job.user is None: 

31 raise ValueError(f"Job {job.pk} has no user, and the change log must name the user of each change.") 

32 # A Django request has each attribute that NetBox copies when an event rule acts on a change. 

33 request = HttpRequest() 

34 request.method = "POST" 

35 request.user = job.user 

36 request.id = job.job_id 

37 # The copy discards what the processors set; before NetBox 4.7, event_tracking keeps it when the body raises. 

38 return contextvars.copy_context().run(_run_under_request_processors, request, body)
