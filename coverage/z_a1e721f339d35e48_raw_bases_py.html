3"""Recover the raw template name a module rule reads as ``{base}``. 

4 

5NetBox keeps no link from an interface to the template that created it, so a rule that reads 

6``{base}`` claims each live name for the template whose raw name it is, or whose raw name the rule 

7renames to it. A name that no single template claims has no base, and nothing renames it.

67def _renamed_pattern(template, variables, raw, historical): 

68 """Return a matcher for the name *template* gives *raw*, at any virtual-chassis position, or None.""" 

69 marked = _marked_evaluation(template, variables, raw) 

70 if marked is None: 

71 return None 

72 groups = { 

73 _MARKERS["base"]: ( 

74 "base", 

75 re.escape(raw) if historical is None else f"(?:{re.escape(raw)}|{historical.pattern})", 

76 ), 

77 _MARKERS["vc_position"]: ("vc", VC_POSITION_DIGITS), 

78 } 

79 pattern = [] 

80 opened = set() 

81 for part in re.split(f"({_MARKERS['base']}|{_MARKERS['vc_position']})", marked): 

82 if part not in groups: 

83 pattern.append(re.escape(part)) 

84 continue 

85 name, alternatives = groups[part] 

86 pattern.append(f"(?P={name})" if name in opened else f"(?P<{name}>{alternatives})") 

87 opened.add(name) 

88 return re.compile("".join(pattern))

91class RawBases: 

92 """The raw template name each live interface name stands for under one module rule. 

93 

94 *families* maps each of the module's interface names outside any channel to its channels, as 

95 ``(name, channel_id)`` pairs. A rule that does not read ``{base}`` takes every name as its own 

96 base, so its claims are never computed. So does a module type without interface templates: it 

97 has no raw names to recover. *catalog* is shared with the family planners, so the module's 

98 templates load once. 

99 """ 

100 

101 def __init__(self, module, rule, variables, families, catalog): 

102 self._module = module 

103 self._rule = rule 

104 self._variables = variables 

105 self._families = dict(families) 

106 self._names = tuple(self._families) 

107 self.catalog = catalog 

108 self._reads_base = rule_reads_base(rule) 

109 self._claimed = False 

110 self._by_name = None 

111 self._ambiguous = frozenset() 

112 

113 def base_for(self, name): 

114 """Return the raw template name *name* stands for, or None when no single template claims it.""" 

115 if not self._reads_base: 

116 return name 

117 self._load() 

118 if self._by_name is None: 

119 return name 

120 return self._by_name.get(name)

122 def is_ambiguous(self, name): 

123 """Return whether more than one template claims *name*, or its template claims another name too.""" 

124 if not self._reads_base: 

125 return False 

126 self._load() 

127 return name in self._ambiguous

129 def _load(self): 

130 """Compute the claims on first use.""" 

131 if not self._claimed: 

132 self._by_name, self._ambiguous = self._claim() 

133 self._claimed = True

135 def _claim(self): 

136 """Return ``(raw name by claimed name, ambiguous names)``, or ``(None, empty)`` without templates. 

137 

138 A template claims its raw name, its historical raw forms and the names the rule gives it. A 

139 raw name beats another template's historical form, as in the drift guard, but never a 

140 renamed form: that overlap is ambiguous, so neither template claims the name. A parent name 

141 without ``{base}`` fits every template, so a template claims it only through a channel the 

142 rule gave that template's base. 

143 """ 

144 claimants = [ 

145 (template.pk, template.template_name, template.resolved, template.historical_pattern) 

146 for template in self.catalog.get() 

147 if template.channel_id is None 

148 ] 

149 if not claimants: 

150 return None, frozenset() 

151 raw_names = {raw for _claimant_id, _template_name, raw, _historical in claimants} 

152 renaming = _renaming_template(self._rule) 

153 needs_channels = _parent_name_needs_channels(self._rule) 

154 claims = [] 

155 raw_by_claimant = {} 

156 for claimant_id, template_name, raw, historical in claimants: 

157 renamed = None if renaming is None else _renamed_pattern(renaming, self._variables, raw, historical) 

158 channels = self._channel_patterns(raw, historical) if needs_channels else None 

159 labels = tuple( 

160 name 

161 for name in self._names 

162 if name == raw 

163 or (historical is not None and name not in raw_names and historical.fullmatch(name)) 

164 or self._is_renamed(name, renamed, channels) 

165 ) 

166 claims.append(TemplateClaim(claimant_id, template_name, labels)) 

167 raw_by_claimant[claimant_id] = raw 

168 accepted, messages = resolve_template_claims(claims, module=self._module, label_kind="raw base") 

169 for message in messages: 

170 logger.warning("%s", message) 

171 by_name = {label: raw_by_claimant[claimant_id] for claimant_id, label in accepted} 

172 return by_name, frozenset(label for claim in claims for label in claim.labels) - by_name.keys()

199class GivenRawNames: 

200 """Bases for names a caller gives as raw template names, such as the names prediction receives. 

201 

202 A given name is its own base, so a stale name is predicted from itself. A name the claim finds 

203 ambiguous has no base, because installation would not rename it either. 

204 """ 

205 

206 def __init__(self, bases): 

207 self._bases = bases 

208 

209 def base_for(self, name): 

210 """Return *name*, or None when the claim over the given names finds it ambiguous.""" 

211 return None if self._bases.is_ambiguous(name) else name
